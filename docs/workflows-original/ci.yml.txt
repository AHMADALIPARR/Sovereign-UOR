name: CI

on:
  push:
    branches: [main]
  pull_request:
    branches: [main]
  workflow_dispatch:

concurrency:
  group: ci-${{ github.ref }}
  cancel-in-progress: true

env:
  SCALA_VERSION: "3.5.2"

jobs:
  # ------------------------------------------------------------------
  # Scala: pure kernel + munit
  # ------------------------------------------------------------------
  scala-test:
    name: Scala Compile & Test
    runs-on: ubuntu-latest
    timeout-minutes: 12
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: "21"
      - name: Install scala-cli
        run: |
          curl -fsSL https://github.com/VirtusLab/scala-cli/releases/latest/download/scala-cli-x86_64-pc-linux.gz \
            | gunzip > /usr/local/bin/scala-cli
          chmod +x /usr/local/bin/scala-cli
      - name: Compile kernel
        run: |
          scala-cli compile --scala ${{ env.SCALA_VERSION }} \
            src/main/scala/uor/kernel/U64.scala \
            src/main/scala/uor/kernel/Poly64.scala
      - name: Unit tests
        run: |
          scala-cli test --scala ${{ env.SCALA_VERSION }} \
            --test-only uor.kernel.Poly64Test \
            src/main/scala/uor/kernel \
            src/test/scala/uor/kernel/Poly64Test.scala

  # ------------------------------------------------------------------
  # Scala: GraalVM native shared library (non-blocking)
  # ------------------------------------------------------------------
  native-image:
    name: Native Image (libuor)
    runs-on: ubuntu-latest
    timeout-minutes: 40
    needs: scala-test
    continue-on-error: true
    steps:
      - uses: actions/checkout@v4
      - uses: graalvm/setup-graalvm@v1
        with:
          java-version: "21"
          distribution: graalvm
          components: "native-image"
          github-token: ${{ secrets.GITHUB_TOKEN }}
          native-image-job-reports: "true"
      - name: Install scala-cli
        run: |
          curl -fsSL https://github.com/VirtusLab/scala-cli/releases/latest/download/scala-cli-x86_64-pc-linux.gz \
            | gunzip > /usr/local/bin/scala-cli
          chmod +x /usr/local/bin/scala-cli
      - name: Compile
        run: |
          mkdir -p build
          scala-cli --power compile --scala ${{ env.SCALA_VERSION }} --server=false -d build \
            src/main/scala/uor/kernel/U64.scala \
            src/main/scala/uor/kernel/Poly64.scala \
            src/main/scala/uor/ffi/UorExports.scala
      - name: native-image --shared
        run: |
          native-image --shared -cp build -H:Name=libuor -O2 --no-fallback \
            -H:+ReportExceptionStackTraces \
            --initialize-at-build-time=uor \
            --initialize-at-build-time=scala.runtime \
            --initialize-at-build-time=java.lang \
            --initialize-at-build-time=java.security \
            -H:+AllowIncompleteClasspath || echo "native-image failed (non-blocking)"
      - name: Verify
        if: success()
        run: |
          ls -lh libuor* 2>/dev/null || true
          file libuor.so || true
      - uses: actions/upload-artifact@v4
        if: success()
        with:
          name: libuor-linux-x86_64
          path: |
            libuor.so
            libuor.h
            graal_isolate*.h
          if-no-files-found: warn
          retention-days: 14

  # ------------------------------------------------------------------
  # montrepol: Cabal build, test, gate
  # ------------------------------------------------------------------
  cabal:
    name: Cabal (montrepol)
    runs-on: ubuntu-24.04
    timeout-minutes: 30
    defaults:
      run:
        working-directory: montrepol
    steps:
      - uses: actions/checkout@v4
      - uses: haskell-actions/setup@v2
        with:
          ghc-version: "9.10.1"
          cabal-version: "3.12.1.0"
      - uses: actions/cache@v4
        with:
          path: |
            ~/.cabal/store
            montrepol/dist-newstyle
          key: ${{ runner.os }}-ghc-9.10.1-${{ hashFiles('montrepol/cabal.project', 'montrepol/packages/*/*.cabal') }}
      - name: cabal update
        run: cabal update || true
      - name: cabal build all
        run: cabal build all
      - name: cabal test all
        run: cabal test all --test-show-details=direct
      - name: cabal run mp-gate
        run: cabal run mp-gate
      - name: CONFORMANCE.md generation check
        run: |
          cabal run mp-gate -- write || true
          git diff --stat CONFORMANCE.md || true
          # Soft check: warn but do not fail the job if formatting differs
          if ! git diff --exit-code CONFORMANCE.md >/dev/null 2>&1; then
            echo "::warning::CONFORMANCE.md differs from generated output"
          fi

  # ------------------------------------------------------------------
  # Final gate (Scala + Cabal required)
  # ------------------------------------------------------------------
  status:
    name: CI Status
    runs-on: ubuntu-latest
    needs: [scala-test, cabal]
    if: always()
    steps:
      - name: Aggregate
        run: |
          echo "Scala tests → ${{ needs.scala-test.result }}"
          echo "Cabal       → ${{ needs.cabal.result }}"
          fail=0
          [[ "${{ needs.scala-test.result }}" == success ]] || fail=1
          [[ "${{ needs.cabal.result }}" == success ]] || fail=1
          if [[ $fail -ne 0 ]]; then
            echo "❌ Scala or Cabal job failed"
            exit 1
          fi
          echo "✅ Scala and Cabal passed"
